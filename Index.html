ชื่อนี้ใช้บันทึกคะแนนและความคืบหน้า

วิธีตั้งค่าบันทึกคะแนนลง Google Sheets (สำหรับครู)

  1. สร้าง Google Sheets ใหม่ ตั้งหัวคอลัมน์แถวแรก: A1: Date B1: Name C1: Score
  2. เมนู ส่วนขยาย (Extensions) > Apps Script แล้วลบโค้ดเดิม วางโค้ดด้านล่าง
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rowData = [];
  var timestamp = Utilities.formatDate(new Date(), "GMT+7", "dd/MM/yyyy HH:mm:ss");
  rowData.push(timestamp);
  rowData.push(e.parameter.name);
  rowData.push(e.parameter.score);
  sheet.appendRow(rowData);
  return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT);
}
  1. กด Deploy > New deployment > เลือก Web App > Who has access = Anyone > Deploy (ยืนยันสิทธิ์บัญชี Google)
  2. คัดลอก Web App URL ไปใส่ในตัวแปร GOOGLE_SCRIPT_URL ในโค้ด HTML
